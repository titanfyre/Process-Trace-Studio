cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ProcessTraceStudio VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022 C++ desktop tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument application, SDK and offline tests with ASAN" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA "/PDBALTPATH:%_PDB%")
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG:FULL)
endif()
add_library(ProcessTraceSDK SDK/ProcessTrace.cpp)
target_link_libraries(ProcessTraceSDK PUBLIC setupapi)
add_executable(ProcessTraceStudio WIN32 App/Main.cpp)
target_link_libraries(ProcessTraceStudio PRIVATE ProcessTraceSDK user32 gdi32 comctl32 comdlg32 shell32)
target_link_options(ProcessTraceStudio PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(ProcessTraceCLI App/Console.cpp)
target_link_libraries(ProcessTraceCLI PRIVATE ProcessTraceSDK)
add_executable(ProcessTraceTests Tests/ProtocolTests.cpp)
target_link_libraries(ProcessTraceTests PRIVATE ProcessTraceSDK)
enable_testing()
add_test(NAME WireRingPrivacyAndPidReuse COMMAND ProcessTraceTests)
add_test(NAME OfflineCsvPreview COMMAND ProcessTraceCLI --preview)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Signed opaque TitanProcessTrace INF/CAT/SYS package, optional")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanProcessTrace.inf TitanProcessTrace.cat TitanProcessTrace.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Driver package is incomplete: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanProcessTrace.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanProcessTrace.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanProcessTrace.sys\"\n")
endif()
add_library(ProcessTraceSetupCore Installer/Setup.cpp)
target_link_libraries(ProcessTraceSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(ProcessTracePackageTests Tests/PackageTests.cpp)
target_include_directories(ProcessTracePackageTests PRIVATE Installer)
target_link_libraries(ProcessTracePackageTests PRIVATE ProcessTraceSetupCore)
add_test(NAME InstallerIdentityAndSignatureGuards COMMAND ProcessTracePackageTests)
add_executable(ProcessTraceSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(ProcessTraceSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target ProcessTraceSetup ProcessTraceSetupConsole)
 target_link_libraries(${target} PRIVATE ProcessTraceSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(target ProcessTraceStudio ProcessTraceCLI ProcessTraceTests ProcessTracePackageTests ProcessTraceSetup ProcessTraceSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
